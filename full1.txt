/**
 * Standard Production Logger
 */
class Logger {
  static prefix = "[Lexicon]";
  static info(msg, ...args) { console.log(`${this.prefix} INFO: ${msg}`, ...args); }
  static warn(msg, ...args) { console.warn(`${this.prefix} WARN: ${msg}`, ...args); }
  static error(msg, ...args) { console.error(`${this.prefix} ERROR: ${msg}`, ...args); }
  static time(label) { console.time(`${this.prefix} ${label}`); }
  static timeEnd(label) { console.timeEnd(`${this.prefix} ${label}`); }
}

/**
 * FNV-1a 32-bit stable hash implementation.
 * @param {string} str
 * @returns {number} 32-bit unsigned integer
 */
function stableHash(str) {
  let h = 2166136261;
  for (let i = 0; i < str.length; i++) {
    h ^= str.charCodeAt(i);
    h = Math.imul(h, 16777619);
  }
  return h >>> 0;
}

/**
 * Deterministically pick an element from an array based on a seed.
 */
function pick(arr, seed) {
  if (!Array.isArray(arr) || arr.length === 0) return "—";
  const h = stableHash(String(seed));
  return arr[h % arr.length];
}

/**
 * Clamps a number between min and max.
 */
function clamp(v, min, max) { return Math.max(min, Math.min(max, v)); }

/**
 * Converts a number to Roman numerals (I - MMMCMXCIX).
 * @param {number} n
 * @returns {string}
 */
function toRoman(n) {
  if (n <= 0 || n > 3999) return String(n);
  const map = [
    [1000, "M"], [900, "CM"], [500, "D"], [400, "CD"],
    [100, "C"], [90, "XC"], [50, "L"], [40, "XL"],
    [10, "X"], [9, "IX"], [5, "V"], [4, "IV"], [1, "I"]
  ];
  let r = "";
  let num = n;
  for (const [val, sym] of map) {
    while (num >= val) { r += sym; num -= val; }
  }
  return r;
}

// ─── Semantic Engine ──────────────────────────────────────

/**
 * Core NLP engine for analyzing text and building a semantic graph.
 */
class SemanticEngine {
  constructor() {
    this.stopWords = new Set([
      "the","a","an","is","are","was","were","be","been","being","have","has","had",
      "do","does","did","will","would","could","should","may","might","must","shall",
      "can","to","of","in","for","on","with","at","by","from","as","into","through",
      "during","before","after","above","below","between","under","again","then","once",
      "here","there","all","any","both","each","few","more","most","other","some","such",
      "no","nor","not","only","own","same","so","than","too","very","just","and","but",
      "if","or","because","until","while","this","that","these","those","i","me","my",
      "we","our","you","your","he","him","his","she","her","it","its","they","them",
      "their","what","which","who","how","why","when","where","s","t"
    ]);
    this.graph = new Map();
    this.docFreq = new Map();
    this.documents = 0;
    this.totalTokens = 0;
  }

  /**
   * Cleans and tokenizes input text.
   * @param {string} text
   * @returns {string[]}
   */
  tokenize(text) {
    if (!text || typeof text !== "string") return [];

    // Normalize and lower case
    let p = text.toLowerCase().normalize("NFC");

    // Expansion of common contractions
    const contractions = {
      "don't": "do not", "doesn't": "does not", "didn't": "did not",
      "won't": "will not", "can't": "cannot", "isn't": "is not",
      "i'm": "i am", "you're": "you are", "it's": "it is",
      "they're": "they are", "i've": "i have", "i'll": "i will",
      "let's": "let us", "that's": "that is", "there's": "there is",
      "wouldn't": "would not", "couldn't": "could not", "shouldn't": "should not"
    };

    for (const [c, e] of Object.entries(contractions)) {
      const esc = c.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
      p = p.replace(new RegExp("\\b" + esc + "\\b", "gi"), e);
    }

    // Remove punctuation except for hyphens inside words
    p = p.replace(/[^\w\s-]/g, " ")
         .replace(/(?<!\w)-+|-+(?!\w)/g, " "); // Remove hyphens not surrounded by alphanumeric chars

    return p.split(/\s+/)
      .filter(w => w.length > 1 && !this.stopWords.has(w));
  }

  /**
   * Adds text to the semantic graph.
   * @param {string} text
   */
  addText(text) {
    const toks = this.tokenize(text);
    if (!toks.length) return;

    Logger.info(`Adding document with ${toks.length} tokens.`);
    this.documents++;
    this.totalTokens += toks.length;

    const uniq = new Set(toks);
    for (const w of uniq) this.docFreq.set(w, (this.docFreq.get(w) || 0) + 1);

    for (const w of toks) {
      if (!this.graph.has(w)) {
